Live Job State and Dynamic 1/3 and 2/3 Profit Split Formula function calculateLiveAccounting() { let gross = 0; let crewWages = 0; let adminProfit = 0; const rows = document.querySelectorAll("table tbody tr[data-price]"); rows.forEach(row => { const price = parseFloat(row.getAttribute("data-price")); const status = row.querySelector(".status-select").value; if (status === "paid") { gross += price; // Exact structured pricing rules if (price === 25) { crewWages += 15; adminProfit += 10; } else if (price === 35) { crewWages += 20; adminProfit += 15; } else if (price === 45) { crewWages += 30; adminProfit += 15; } else { // Standard fallback fallback logic crewWages += (price * 2) / 3; adminProfit += price / 3; } } }); document.getElementById("gross-val").innerText = "$" + gross.toFixed(2); document.getElementById("crew-val").innerText = "$" + crewWages.toFixed(2); document.getElementById("net-val").innerText = "$" + adminProfit.toFixed(2); } // 2. Predictive Engine & Live Milestone Tracker Calculations function updatePredictor() { const crews = parseInt(document.getElementById("pred-workers").value) || 0; const yards = parseInt(document.getElementById("pred-yards").value) || 0; const weeks = parseInt(document.getElementById("pred-weeks").value) || 0; const totalYardsProjected = crews * yards * weeks; const target = 250; let percentage = (totalYardsProjected / target) * 100; if (percentage > 100) percentage = 100; if (percentage < 0) percentage = 0; document.getElementById("progress-bar").style.width = percentage + "%"; document.getElementById("progress-text").innerText = "Current Progress Level: " + Math.round(percentage) + "% Complete (" + totalYardsProjected + " Yards Projected)"; } // 3. Disciplinary Action Systems function deductCrew(crewName) { let amount = prompt("Enter payment deduction fine amount for " + crewName + " ($):", "5.00"); if (amount != null && amount !== "") { alert("Applied code deduction fine of $" + parseFloat(amount).toFixed(2) + " to " + crewName + "'s layout balance."); } } function suspendCrew(crewName, rowId) { const statusCell = document.querySelector("#" + rowId + " .crew-status"); statusCell.innerHTML = 'Suspended'; alert(crewName + " has been temporarily locked out of active scheduling queues."); } function kickCrew(crewName, rowId) { if (confirm("Are you absolutely sure you want to kick and terminate " + crewName + " from operations?")) { document.getElementById(rowId).remove(); alert(crewName + " removed from system registry successfully."); } } // Run updates instantly on window boot up window.onload = function() { calculateLiveAccounting(); updatePredictor(); };
